```python
    {"role": "user", "content": prompt}

# Функция очистки приветствий
def _remove_greetings(self, answer):
    greeting_patterns = [
        r'^Привет[^!]*!\s*',
        r'^Здравствуй[^!]*!\s*',
        r'^Добро пожаловать[^!]*!\s*'
    ]
    for pattern in greeting_patterns:
        answer = re.sub(pattern, '', answer, flags=re.IGNORECASE)
    return answer.strip()
```
